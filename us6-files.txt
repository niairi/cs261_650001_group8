const OPEN = '09:00';   
const CLOSE = '17:00';  
const MAX_HOURS = 3;    

const DATE_RE = /^\d{4}-\d{2}-\d{2}$/;
const TIME_RE = /^([01]\d|2[0-3]):00$/; 

const toMinutes = (t) => {
  const [h, m] = t.split(':').map(Number);
  return h * 60 + m;
};

const isRealDate = (d) =>
  DATE_RE.test(d) && new Date(`${d}T00:00:00Z`).toISOString().slice(0, 10) === d;


function validateBooking(body) {
  const errors = [];
  const roomId = Number(body.room_id);
  const people = Number(body.people);
  const { booking_date, start_time, end_time } = body;
  const purpose = typeof body.purpose === 'string' && body.purpose.trim() ? body.purpose.trim() : null;

  if (!Number.isInteger(roomId) || roomId <= 0) errors.push('room_id ต้องเป็นตัวเลข');
  if (!Number.isInteger(people) || people < 1) errors.push('จำนวนผู้เข้าใช้ต้องเป็นจำนวนเต็มตั้งแต่ 1 คนขึ้นไป');
  if (typeof booking_date !== 'string' || !isRealDate(booking_date)) errors.push('วันที่ต้องอยู่ในรูปแบบ YYYY-MM-DD');
  if (typeof start_time !== 'string' || !TIME_RE.test(start_time)) errors.push('เวลาเริ่มต้องเป็นเวลาเต็มชั่วโมง เช่น 09:00');
  if (typeof end_time !== 'string' || !TIME_RE.test(end_time)) errors.push('เวลาสิ้นสุดต้องเป็นเวลาเต็มชั่วโมง เช่น 12:00');
  if (purpose && purpose.length > 255) errors.push('วัตถุประสงค์ยาวได้ไม่เกิน 255 ตัวอักษร');
  if (errors.length) throw new HttpError(400, 'ข้อมูลการจองไม่ถูกต้อง', errors);

  const start = toMinutes(start_time);
  const end = toMinutes(end_time);
  if (start < toMinutes(OPEN) || end > toMinutes(CLOSE)) errors.push(`จองได้เฉพาะช่วง ${OPEN} - ${CLOSE} น.`);
  if (end <= start) errors.push('เวลาสิ้นสุดต้องหลังเวลาเริ่มต้น');
  if (end - start > MAX_HOURS * 60) errors.push(`จองได้ครั้งละไม่เกิน ${MAX_HOURS} ชั่วโมง`);

  // ห้ามจองเวลาที่ผ่านมาแล้ว (ใช้เวลาของเครื่องที่รัน server)
  const now = new Date();
  const today = now.toLocaleDateString('sv-SE'); // ได้รูปแบบ YYYY-MM-DD
  const nowTime = now.toTimeString().slice(0, 5); // HH:MM
  if (booking_date < today || (booking_date === today && start_time <= nowTime)) {
    errors.push('ไม่สามารถจองเวลาที่ผ่านมาแล้ว');
  }

  // จองล่วงหน้าได้ไม่เกิน BOOKING_DAYS_AHEAD วัน (ค่าเริ่มต้น 7 วัน นับวันนี้เป็นวันที่ 0)
  const MAX_DAYS_AHEAD = Number(process.env.BOOKING_DAYS_AHEAD || 7);
  const last = new Date(now);
  last.setDate(last.getDate() + MAX_DAYS_AHEAD);
  if (booking_date > last.toLocaleDateString('sv-SE')) {
    errors.push(`จองล่วงหน้าได้ไม่เกิน ${MAX_DAYS_AHEAD} วัน`);
  }
  if (errors.length) throw new HttpError(400, 'ข้อมูลการจองไม่ถูกต้อง', errors);

  return { roomId, people, purpose, booking_date, start_time, end_time };
}

         FROM booking
        WHERE booking_date = $1

        ORDER BY room_id, start_time`,
      [date]
    );

    next(err);
  }
});

router.patch('/:id/cancel', requireAuth, async (req, res, next) => {
  let client;
  try {
    const bookingId = Number(req.params.id);
    if (!Number.isInteger(bookingId) || bookingId <= 0) {
      throw new HttpError(400, 'รหัสการจองไม่ถูกต้อง');
    }

    client = await pool.connect();
    await client.query('BEGIN');

    // FOR UPDATE กันการยกเลิกซ้อนกันพร้อมกัน
    const found = await client.query(
      `SELECT booking_id, student_id, status,
              to_char(booking_date, 'YYYY-MM-DD') AS booking_date,
              to_char(start_time, 'HH24:MI') AS start_time
         FROM booking
        WHERE booking_id = $1
        FOR UPDATE`,
      [bookingId]
    );
    if (found.rows.length === 0) throw new HttpError(404, 'ไม่พบการจองนี้');

    const booking = found.rows[0];
    if (booking.student_id !== req.user.student_id) {
      throw new HttpError(403, 'ยกเลิกได้เฉพาะการจองของตัวเองเท่านั้น');
    }
    if (booking.status === 'CANCELLED') throw new HttpError(409, 'การจองนี้ถูกยกเลิกไปแล้ว');
    if (booking.status === 'EXPIRED') throw new HttpError(409, 'การจองนี้หมดอายุแล้ว');

    // ยกเลิกได้ก่อนเวลาเริ่มอย่างน้อย 30 นาที (US4 TC4)
       const CANCEL_BEFORE_MIN = 30;
       const startAt = new Date(`${booking.booking_date}T${booking.start_time}:00`);
       if (startAt.getTime() - Date.now() < CANCEL_BEFORE_MIN * 60 * 1000) {
         throw new HttpError(400, `ยกเลิกได้ก่อนเวลาเริ่มอย่างน้อย ${CANCEL_BEFORE_MIN} นาที`);
       }

    // เปลี่ยนสถานะแทนการลบ เพื่อเก็บประวัติไว้
    // JOIN room เพื่อได้ชื่อห้องไปใส่ในอีเมล
    const updated = await client.query(
      `UPDATE booking b
          SET status = 'CANCELLED'
         FROM room r
        WHERE b.booking_id = $1
          AND r.room_id = b.room_id
        RETURNING b.booking_id, b.student_id, b.room_id, r.room_name,
                  to_char(b.booking_date, 'YYYY-MM-DD') AS booking_date,
                  to_char(b.start_time, 'HH24:MI') AS start_time,
                  to_char(b.end_time, 'HH24:MI') AS end_time,
                  b.status`,
      [bookingId]
    );

    await client.query('COMMIT');

    const row = updated.rows[0];
    // ส่งอีเมลยืนยันการยกเลิก (US7) — ไม่รอ และถ้าส่งไม่ได้ก็ไม่ทำให้การยกเลิกล้มเหลว
    sendBookingCancellation(req.user.email, row)
      .catch((err) => console.error('ส่งอีเมลยืนยันการยกเลิกไม่สำเร็จ:', err.message));
    res.json({
      message: 'ยกเลิกการจองสำเร็จ',
      booking: { ...row, booking_id: Number(row.booking_id), room_id: Number(row.room_id) },
    });
  } catch (err) {
    if (client) await client.query('ROLLBACK').catch(() => {});
    next(err);
  } finally {
    if (client) client.release();
  }
});

let transporter; // สร้างครั้งเดียวแล้วใช้ซ้ำ

function getTransporter() {
  if (transporter !== undefined) return transporter;
  if (!process.env.SMTP_HOST) {
    transporter = null;
    return null;
  }
  const nodemailer = require('nodemailer');
  const port = Number(process.env.SMTP_PORT || 587);
  transporter = nodemailer.createTransport({
    host: process.env.SMTP_HOST,
    port,
    secure: port === 465,
    auth: { user: process.env.SMTP_USER, pass: process.env.SMTP_PASS },
  });
  return transporter;
}

// "2026-10-20" -> "วันอังคารที่ 20 ตุลาคม 2569"
function formatThaiDate(date) {
  return new Date(`${date}T00:00:00+07:00`).toLocaleDateString('th-TH', {
    timeZone: 'Asia/Bangkok',
    weekday: 'long',
    year: 'numeric',
    month: 'long',
    day: 'numeric',
  });
}

const escapeHtml = (s) =>
  String(s).replace(/[&<>"']/g, (c) => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c]));

// หัวข้ออีเมลตาม US5 TC1/TC2 เช่น "[CS Co-Working] ยืนยันการจอง Room A (101) 22/10/2569 10:00–12:00"
function emailSubject(action, booking) {
  const shortDate = new Date(`${booking.booking_date}T00:00:00+07:00`).toLocaleDateString('th-TH', {
    timeZone: 'Asia/Bangkok',
    day: '2-digit',
    month: '2-digit',
    year: 'numeric',
  });
  const roomLabel = booking.room_id ? `${booking.room_name} (${booking.room_id})` : booking.room_name;
  return `[CS Co-Working] ${action} ${roomLabel} ${shortDate} ${booking.start_time}–${booking.end_time}`;
}

// บันทึกผลการส่งลงตาราง email_logs (Task32) — ถ้าบันทึกไม่ได้ไม่ให้กระทบการส่งอีเมล
async function defaultLog(entry) {
  const pool = require('../db');
  const { insertEmailLog } = require('../repositories/emailLogRepo');
  await insertEmailLog(pool, entry);
}

async function safeLog(log, entry) {
  try {
    await log(entry);
  } catch (err) {
    console.error('[อีเมล] บันทึก email_logs ไม่สำเร็จ:', err.message);
  }
}

// ส่งอีเมลเกี่ยวกับการจอง 1 ฉบับ — คืนค่า { sent: true/false, reason }
// type = BOOKING_CONFIRMED | BOOKING_CANCELLED (บันทึกลง email_logs.email_type)
async function sendBookingEmail(type, mail, to, booking, transport, log) {
  const base = { booking_id: booking.booking_id, to_email: to || null, email_type: type };

  if (!to) {
    console.warn(`[อีเมล] ไม่มีอีเมลของผู้จอง (booking #${booking.booking_id}) จึงไม่ได้ส่ง`);
    await safeLog(log, { ...base, status: 'SKIPPED', error_message: 'no-recipient' });
    return { sent: false, reason: 'no-recipient' };
  }
  if (!transport) {
    console.log(`[อีเมล] ยังไม่ได้ตั้ง SMTP จึงแสดงแทนการส่ง\nถึง: ${to}\nเรื่อง: ${mail.subject}\n${mail.text}\n`);
    await safeLog(log, { ...base, status: 'SKIPPED', error_message: 'smtp-not-configured' });
    return { sent: false, reason: 'smtp-not-configured' };
  }
  try {
    await transport.sendMail({
      from: process.env.MAIL_FROM || process.env.SMTP_USER,
      to,
      subject: mail.subject,
      text: mail.text,
      html: mail.html,
    });
  } catch (err) {
    await safeLog(log, { ...base, status: 'FAILED', error_message: err.message });
    throw err;
  }
  await safeLog(log, { ...base, status: 'SENT' });
  return { sent: true };
}

// อีเมลยืนยันการจอง — transport และ log ส่งเข้ามาเองได้ (ใช้ตอน test)
function sendBookingConfirmation(to, booking, transport = getTransporter(), log = defaultLog) {
  return sendBookingEmail('BOOKING_CONFIRMED', buildBookingConfirmationEmail(booking), to, booking, transport, log);

// อ่าน/บันทึกตาราง users (Task28)

// บันทึกผู้ใช้ตอน login: ถ้ายังไม่มีให้เพิ่ม ถ้ามีแล้วให้อัปเดตข้อมูลล่าสุดจาก TU API
async function upsertUser(db, user) {
  const { rows } = await db.query(
    `INSERT INTO users (student_id, name_th, name_en, email, type, department, faculty, role)
     VALUES ($1, $2, $3, $4, $5, $6, $7, $8)
     ON CONFLICT (student_id) DO UPDATE SET
       name_th = EXCLUDED.name_th,
       name_en = EXCLUDED.name_en,
       email = EXCLUDED.email,
       type = EXCLUDED.type,
       department = EXCLUDED.department,
       faculty = EXCLUDED.faculty,
       role = EXCLUDED.role,
       last_login_at = CURRENT_TIMESTAMP
     RETURNING student_id, created_at, last_login_at`,
    [user.student_id, user.name_th, user.name_en, user.email, user.type, user.department, user.faculty, user.role]
  );
  return rows[0];
}

===== database/schema.sql =====
CREATE TABLE room (
    room_id BIGSERIAL PRIMARY KEY,
    room_name VARCHAR(100) NOT NULL,
    capacity INTEGER NOT NULL CHECK (capacity > 0),
    location VARCHAR(255),
    is_active BOOLEAN NOT NULL DEFAULT TRUE
);

CREATE TABLE booking (
    booking_id BIGSERIAL PRIMARY KEY,
    student_id VARCHAR(50) NOT NULL,
    room_id BIGINT NOT NULL,
    booking_date DATE NOT NULL,
    start_time TIME NOT NULL,
    end_time TIME NOT NULL,
    people INTEGER NOT NULL CHECK (people > 0),
    purpose VARCHAR(255),
    status VARCHAR(20) NOT NULL DEFAULT 'BOOKED',
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    confirmed_at TIMESTAMP,

    FOREIGN KEY (room_id) REFERENCES room(room_id),

    CHECK (end_time > start_time),
    CHECK (end_time - start_time <= INTERVAL '3 hours'),
    CHECK (end_time <= TIME '17:00'),
    CHECK (status IN ('BOOKED', 'CONFIRMED', 'CANCELLED', 'EXPIRED'))
);

-- Booking can be made up to 7 days in advance
CREATE OR REPLACE FUNCTION check_booking_day_ahead()
RETURNS TRIGGER AS $$
BEGIN
    IF NEW.booking_date > CURRENT_DATE + 7 THEN
        RAISE EXCEPTION 'Booking date cannot be more than 7 days in advance';
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_check_booking_day_ahead
BEFORE INSERT OR UPDATE OF booking_date ON booking
FOR EACH ROW
EXECUTE FUNCTION check_booking_day_ahead();
